# $denyallow, which the engine ignores, rewritten as a block filter plus exceptions. From uBO's list:
#   $script,3p,denyallow=bootstrapcdn.com,domain=uploadflix.*
block https://cdn.ad-network.example/x.js https://uploadflix.org/ script
allow https://maxcdn.bootstrapcdn.com/bootstrap/4.0.0/js/bootstrap.min.js https://uploadflix.org/ script

# Generic cosmetic filters apply when the page has a matching class or id.
hide https://www.example.com/ ads-banner - .ads-banner

show https://www.example.com/ - - .ads-banner

# uBO's generic exception #@#.reklama turns .reklama hiding off on every page. The engine rejects generic exceptions,
